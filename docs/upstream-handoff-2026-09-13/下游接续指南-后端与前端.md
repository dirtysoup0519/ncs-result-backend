## 2. 路线 A：Flask 直接读 CSV（最快出屏）

node100 上装 Flask：

```bash
pip install flask flask-cors
```

`app.py`（最小可用版，8 个接口）：

```python
# -*- coding: utf-8 -*-
from flask import Flask, jsonify
from flask_cors import CORS
import csv, os

app = Flask(__name__)
CORS(app)  # 允许前端跨域访问

DATA = "/home/hadoop/dashboard/data"

def read_tsv(name, cols):
    rows = []
    with open(os.path.join(DATA, name), encoding="utf-8") as f:
        for line in f:
            parts = line.rstrip("\n").split("\t")
            rows.append(dict(zip(cols, parts)))
    return rows

@app.route("/api/kpi")
def kpi():
    cols = ["total_order_cnt","total_fees","total_kwh","total_user_cnt",
            "total_station_cnt","avg_fee_per_order","avg_kwh_per_order"]
    return jsonify(read_tsv("kpi_total.csv", cols)[0])

@app.route("/api/revenue_trend")
def revenue_trend():
    cols = ["stat_date","order_cnt","total_fees","total_kwh"]
    return jsonify(read_tsv("revenue_trend.csv", cols))

@app.route("/api/revenue_monthly")
def revenue_monthly():
    cols = ["stat_month","order_cnt","total_fees","total_kwh"]
    return jsonify(read_tsv("revenue_monthly.csv", cols))

@app.route("/api/station_top10")
def station_top10():
    cols = ["station_id","station_name","location_id","order_cnt","total_fees","total_kwh"]
    return jsonify(read_tsv("station_top10.csv", cols))

@app.route("/api/platform_stat")
def platform_stat():
    cols = ["platform","order_cnt","total_fees","fee_ratio"]
    return jsonify(read_tsv("platform_stat.csv", cols))

@app.route("/api/hourly_stat")
def hourly_stat():
    cols = ["stat_hour","order_cnt","total_kwh","total_fees"]
    return jsonify(read_tsv("hourly_stat.csv", cols))

@app.route("/api/order_daily")
def order_daily():
    cols = ["stat_date","order_cnt","total_kwh","total_fees","total_charge_hours","user_cnt"]
    return jsonify(read_tsv("order_daily.csv", cols))

@app.route("/api/process_daily")
def process_daily():
    cols = ["stat_date","record_cnt","session_cnt","avg_soc","max_soc","min_soc",
            "avg_current","avg_pack_voltage","avg_max_temp"]
    return jsonify(read_tsv("process_daily.csv", cols))

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
```

启动：

```bash
python app.py
# 浏览器验证：http://192.168.176.100:5000/api/kpi
```
